### 阶段 B：打包配置页 / 运行页双态权限控制

**工作量**：新增1个打包配置HTML + 修改3个index.html的权限逻辑
**风险**：中

#### B1. 设计思路

```
打包流程：
  打包者运行 build.bat → 弹出打包配置页面 → 编辑诊所名/医师名 → 保存到config.json → 打包

运行流程：
  用户安装后启动 → 读取config.json → 诊所名/医师名显示为只读
```

#### B2. 新增打包配置页面

创建 `shared/packaging-config.html`：
- 读取当前目录的 config.json
- 诊所名称输入框（可编辑）
- 电脑个人定制版：额外显示医师姓名输入框（可编辑）
- 电脑诊所定制版：仅诊所名称可编辑
- 保存按钮 → 写入 config.json → 继续打包流程

#### B3. 修改 build.bat 集成配置页

```batch
@echo off
:: 打包前先打开配置页面
start "" "%~dp0packaging-config.html"
:: 等待配置完成（检测config.json的修改时间）
:: 然后继续打包流程
```

#### B4. 运行页权限控制

在 index.html 中新增版本权限控制函数：

```javascript
// 根据 edition 控制字段读写权限
function applyEditionPermissions(edition) {
    const clinicNameInput = document.getElementById('clinicName');
    const defaultDoctorInput = document.getElementById('defaultDoctor');
    const doctorNameInput = document.getElementById('doctorName');

    if (edition === 'personal') {
        // 个人定制版：诊所名、医师名全部只读
        [clinicNameInput, defaultDoctorInput, doctorNameInput].forEach(el => {
            if (el) { el.readOnly = true; el.style.backgroundColor = '#f0f0f0'; }
        });
    } else if (edition === 'clinic_custom') {
        // 诊所定制版：诊所名只读，医师名可编辑
        if (clinicNameInput) { clinicNameInput.readOnly = true; clinicNameInput.style.backgroundColor = '#f0f0f0'; }
    }
    // cloud 和 offline 版：全部可编辑，不限制
}
```

| 版本 | 诊所名 | 医师名 | 用户管理 |
|------|--------|--------|---------|
| 电脑个人定制版 | 运行页只读 | 运行页只读 | 隐藏 |
| 电脑诊所定制版 | 运行页只读 | 可编辑 | 可增删改 |
| 电脑云端诊所版 | 可编辑 | 可编辑 | 可增删改 |
| 电脑离线诊所版 | 可编辑 | 可编辑 | 可增删改 |

#### E3. version-config.js 设计

```javascript
// shared/version-config.js
const VersionConfig = {
    _edition: null,
    _config: null,

    async init() {
        // 从 config.json 或 localStorage 读取版本信息
        if (window.electronAPI) {
            this._config = await window.electronAPI.getAppConfig();
        } else {
            const stored = localStorage.getItem('app_config');
            this._config = stored ? JSON.parse(stored) : {};
        }
        this._edition = this._config.edition || 'cloud';
    },

    get edition() { return this._edition; },
    get isCloud() { return this._edition === 'cloud'; },
    get isOffline() { return this._edition === 'offline'; },
    get isPersonal() { return this._edition === 'personal'; },
    get isClinicCustom() { return this._edition === 'clinic_custom'; },
    get versionLabel() { return this._config.versionLabel || '未知版本'; },
    get productName() { return this._config.productName || '惠康中医诊所管理系统'; },

    // 权限判断
    canEditClinicName() { return this.isCloud() || this.isOffline(); },
    canEditDoctorName() { return !this.isPersonal(); },
    canManageUsers() { return !this.isPersonal(); },
    canSync() { return this.isCloud(); },
    hasMultiUser() { return !this.isPersonal(); },
    hasRememberPassword() { return this.isPersonal(); }
};
```

// shared/debug-logger.js
const DBG = {
    _enabled: location.search.includes('debug') || localStorage.getItem('debug') === '1',

    log(module, action, data) {
        if (!this._enabled) return;
        const version = window.VersionConfig ? VersionConfig.versionLabel : 'unknown';
        const device = navigator.userAgent.includes('Android') ? 'Android' : 'Desktop';
        console.log(`[DBG][${version}][${device}][${module}] ${action}`, data || '');
    },

};
